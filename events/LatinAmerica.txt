	id = latin.10
	title = latin.10.t
	desc = latin.10.d
	picture = GFX_guarrillas
	fire_only_once = yes
	trigger = {
		AND = {
			num_of_controlled_states < 2
			has_idea = gaulle 
			has_country_flag = latin_america
		}
		NOT = {
			OR = {
				has_government = communism
				tag = CAN
				tag = USA
				tag = CUB
			}
		}
		OR = {
			has_stability < 0.3
			communism > 0.2
		}
	} 

	option = {
		name = latin.10.a
		random_list = {
			55 = {
				news_event = { id = latin.11 }
				complete_national_focus = common_2
				complete_national_focus = common_26
				set_politics = { ruling_party = communism }
			}
			45 = {
				country_event = { id = latin.12 }
			}
		}
	}

	option = {
		name = latin.10.b
			news_event = { id = latin.11 }
			complete_national_focus = common_2
			complete_national_focus = common_26
			set_politics = { ruling_party = communism }
	}
}
