news_event = {
	id = Olympic.1948
	title = Olympic.1948.t
	desc = Olympic.1948.d
	picture = GFX_1948LondonOlympic
	major = yes
	trigger = {
		date > 1948.7.1
		date < 1948.9.15
		country_exists = ENG
		ENG = { has_idea = Olympic }
		tag = ENG
	}
	mean_time_to_happen = {
		days = 2
	}
	option = {
		name = Olympic.66.a
		if = { 
			limit = { tag = ENG }
			remove_ideas = Olympic
		}
		if = { 
			limit = { tag = AST }
			add_ideas = Olympic
		}
	}
}
