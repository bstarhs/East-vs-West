
add_namespace = spyaction
country_event = { 
	id = spyaction.1 
	hidden = yes
	fire_only_once = yes
	mean_time_to_happen = { days = 100 }
	trigger = {
		any_country = {
			network_national_coverage = {
				target = ROOT
				value > 10
			}
		}
		OR = {
			has_tech = atomic_research
			is_researching_technology = atomic_research
			is_researching_technology = nukes
		}
	}
	immediate = {
		random_country = {
			limit = {
				network_national_coverage = {
					target = ROOT
					value > 10
				}
			}
			country_event = { id = spyaction.2  }
		}
	}
	option = {
		name = news.51.a
	}
}
