#电力计算
news_event = {
	id = count.1
	hidden = yes
	mean_time_to_happen = {
		days = 10
	}
	trigger = {
		has_tech = basic_machine_tools
		any_state = {
			OR = {
				power_plant > 0
				dam_plant > 0
			}
		}
	}
	option = {
		name = news.51.a
			set_variable = { power = 0 }
			every_owned_state = {
				if = {
					limit = {
						check_variable = { this.state_power_need > 0 }
					}
					set_variable = { this.state_power_need = 0 }
					set_variable = { this.heavy_industry_need = 0 }
				}
			}
			every_owned_state = {
				if = {
					limit = {
						check_variable = { this.state_power_production > 0 }
					}
					set_variable = { this.state_power_production = 0 }
					set_variable = { this.plant_power = 0 }
				}
			}
			every_owned_state = {
				#计算电力需求
				if = {
					limit = {
						OR = {
							has_state_category = pastoral
							has_state_category = rural
							has_state_category = town
							has_state_category = large_town
						}
						owner = { has_tech = dispersed_industry4 }
					}
					add_to_variable = { state_power_need = 2 }
				}
			}
			every_owned_state = {
				if = {
					limit = { has_state_category = rural }
					add_to_variable = { state_power_need = 1 }
				}
				if = {
					limit = { has_state_category = town }
					add_to_variable = { state_power_need = 2 }
				}
				if = {
					limit = { has_state_category = large_town }
					add_to_variable = { state_power_need = 4 }
				}
				if = {
					limit = { has_state_category = city }
					add_to_variable = { state_power_need = 8 }
				}
				if = {
					limit = { has_state_category = large_city }
					add_to_variable = { state_power_need = 10 }
				}
				if = {
					limit = { has_state_category = metropolis }
					add_to_variable = { state_power_need = 10 }
				}
				if = {
					limit = { has_state_category = megalopolis }
					add_to_variable = { state_power_need = 12 }
				}
			}
			every_owned_state = {
				if = {
					limit = { owner = { has_tech = advanced_machine_tools } }
					add_to_variable = { heavy_industry_need = building_level@industrial_complex }
				}
			}
			every_owned_state = {
				if = {
					limit = {
						OR = {
							synthetic_refinery > 0
							steel_making > 0
						}
					}
					add_to_variable = { heavy_industry_need = building_level@synthetic_refinery }
					add_to_variable = { heavy_industry_need = building_level@steel_making }
				}
				add_to_variable = { state_power_need = heavy_industry_need }
			}
			##发电
			every_owned_state = {
				if = {
					limit = {
						owner = {
							has_tech = power_generation1
							has_fuel > 400
							has_dynamic_modifier = { modifier = power_general_fuel }
						}
					}
					add_to_variable = { this.plant_power = this.resource@power }
					add_to_variable = { this.plant_power = this.building_level@power_plant }
				}
				if = {
					limit = {
						owner = {
							has_tech = power_generation1
							check_variable = { resource@coal < 1 } 
							NOT = { has_dynamic_modifier = { modifier = power_general_fuel } }
						}
					}
					
					add_to_variable = { this.plant_power = this.resource@power }
					multiply_variable = { this.plant_power = 0.3 }
				}
				if = {
					limit = {
						owner = {
							has_tech = power_generation1
							check_variable = { resource@coal > 0 } 
							NOT = { has_dynamic_modifier = { modifier = power_general_fuel } }
						}
					}
					add_to_variable = { this.plant_power = this.resource@power }
				}
			}
			#核能发电
			if = {
				limit = {
					has_tech = nuclear_plant_1
					chromium > 4
					
				}
				every_owned_state = {
					limit = {
						check_variable = { this.building_level@nuclear_reactor > 0 }
					}
					add_to_variable = { this.plant_power = this.building_level@nuclear_reactor }
					add_to_variable = { this.plant_power = this.building_level@nuclear_reactor }
					add_to_variable = { this.plant_power = this.building_level@nuclear_reactor }
					add_to_variable = { this.plant_power = this.building_level@nuclear_reactor }
					add_to_variable = { this.plant_power = this.building_level@nuclear_reactor }
					add_to_variable = { this.plant_power = this.building_level@nuclear_reactor }
					add_to_variable = { this.plant_power = this.building_level@nuclear_reactor }
					add_to_variable = { this.plant_power = this.building_level@nuclear_reactor }
				}
			}
			#电力相邻补充
			every_owned_state = {
				if = {
					limit = {
						check_variable = { this.plant_power < state_power_need }
						check_variable = { this.infrastructure_level > 1 }
						owner = {
							has_tech = construction2
						}
					}
					set_variable = { this.state_power_neighber = 0 }
					every_neighbor_state = {
						if = {
							limit = {
								check_variable = { this.building_level@nuclear_reactor < 1 }
								check_variable = { this.plant_power > this.state_power_need }
								check_variable = { THIS.controller = PREV.controller }
							}
							add_to_variable = { PREV.state_power_neighber = 0.6 }
						}
					}
					every_neighbor_state = {
						if = {
							limit = {
								check_variable = { this.building_level@nuclear_reactor > 0 }
								check_variable = { this.plant_power > this.state_power_need }
								check_variable = { THIS.controller = PREV.controller }
							}
							add_to_variable = { PREV.state_power_neighber = building_level@nuclear_reactor }
							add_to_variable = { PREV.state_power_neighber = building_level@nuclear_reactor }
							add_to_variable = { PREV.state_power_neighber = building_level@nuclear_reactor }
						}
					}
				}
			}
			#省份总电力产出
			every_owned_state = {
				add_to_variable = { this.state_power_production = this.plant_power }
				add_to_variable = { this.state_power_production = this.state_power_neighber }
			}
			#消耗煤炭
			set_variable = { this.powerplant_coal_coast = 0 }
			every_owned_state = {
				limit = { check_variable = { this.plant_power > 0} }
				add_to_variable = { prev.powerplant_coal_coast = this.building_level@power_plant }
			}
			multiply_variable = { this.powerplant_coal_coast = 0.5 }
			#满足电力数量
			every_owned_state = {
				limit = {
					OR = {
						check_variable = { state_power_production > state_power_need }
					}
				}
				add_to_variable = { PREV.power = 1 }
			}
	}
}
#粮食计算
news_event = {
	id = count.2
	hidden = yes
	mean_time_to_happen = {
		days = 20
	}
	trigger = {
		has_tech = dispersed_industry
		any_state = {
			farm_building > 0
		}
	}
	option = {
		name = news.51.a
		#粮食需求
		every_owned_state = {
			if = {
				limit = { check_variable = { state_food_need > 0 } }
				set_variable = { state_food_need = 0 }
			}
		}
		every_owned_state = {
				####
				if = {
					limit = { has_state_category = pastoral }
					set_variable = { urban_need = 2 }
				}
				if = {
					limit = { has_state_category = small_island }
					set_variable = { urban_need = 1 }
				}
				if = {
					limit = { has_state_category = rural }
					set_variable = { urban_need = 4 }
				}
				####
				if = {
					limit = { has_state_category = town }
					set_variable = { urban_need = 8 }
				}
				if = {
					limit = { has_state_category = large_town }
					set_variable = { urban_need = 10 }
				}
				####
				if = {
					limit = { has_state_category = city }
					set_variable = { urban_need = 12 }
				}
				if = {
					limit = { has_state_category = large_city }
					set_variable = { urban_need = 14 }
				}
				####
				if = {
					limit = { has_state_category = megalopolis }
					set_variable = { urban_need = 16 }
				}
				####
				if = {
					limit = { has_state_category = metropolis }
					set_variable = { urban_need = 20 }
				}
				####
				if = {
					limit = { has_dynamic_modifier = { modifier = rice_food_production } }
					multiply_variable = { this.urban_need = 2 }
				}
				
			}
			##工业建筑食品需求
			every_owned_state = {
				set_variable = { this.factories_foods_need = 0 }
				add_to_variable = { this.factories_foods_need = this.building_level@industrial_complex }
				add_to_variable = { this.factories_foods_need = this.building_level@arms_factory }
				add_to_variable = { this.factories_foods_need = this.building_level@dockyard }
				add_to_variable = { this.factories_foods_need = this.building_level@power_plant }
				multiply_variable = { this.factories_foods_need = 0.5 }
				add_to_variable = { this.factories_foods_need = this.building_level@synthetic_refinery }
				add_to_variable = { this.factories_foods_need = this.building_level@oilfield }
				add_to_variable = { this.factories_foods_need = this.building_level@steel_making }
				add_to_variable = { this.factories_foods_need = this.building_level@steel_making }
				add_to_variable = { this.factories_foods_need = this.building_level@rubber_factory }
				add_to_variable = { this.factories_foods_need = this.building_level@dam_plant }
			}
			##政策影响食品需求
			if = {
				limit = { has_idea = war_economy }
				every_owned_state = {
					if = {
						limit = { check_variable = { this.factories_foods_need > 0 } }
						multiply_variable = { this.factories_foods_need = 1.5 }
					}
				}
			}
			##磷酸盐加成
			if = {
				limit = {
					has_tech = dispersed_industry2
					check_variable = { phosphate > num_owned_states }
					any_owned_state = {
						NOT = { has_dynamic_modifier = { modifier = state_phosphate_modifiers } }
					}
				}
				every_owned_state = {
					if = {
					limit = {
						NOT = { has_dynamic_modifier = { modifier = state_phosphate_modifiers } }
						}
						add_dynamic_modifier = { modifier = state_phosphate_modifiers }
					}
				}
			}
			##殖民地加成
			if = {
				limit = {
					check_variable = { this.food_from_colonies > 0 }
				}
				add_to_variable = { food_num = food_from_colonies }
			}
			##食品需求总计算
			every_owned_state = {
				add_to_variable = { state_food_need = urban_need }
				add_to_variable = { this.state_food_need = this.factories_foods_need }
				if = {
					limit = { has_dynamic_modifier = { modifier = rice_food_production } }
					multiply_variable = { this.state_food_need = 1.25 }
				}
			}
			##buff影响的食物需求
			if = {
				limit = { has_idea = PeoplesCommuni }
				every_owned_state = { multiply_variable = { state_food_need = 1.5 } }
			}
			if = {
				limit = { has_completed_focus = prc9 }
				every_owned_state = { multiply_variable = { state_food_need = 0.8 } }
			}
			##计算相邻粮食
			every_owned_state = {
				if = {
					limit = { check_variable = { this.state_food_neighber > 0 } }
					add_to_variable = { this.state_food_production = this.state_food_neighber }
				}
			}
			##计算调配粮食
			every_owned_state = {
				if = {
					limit = { check_variable = { state_food_added < 0 } }
					multiply_variable = { state_food_added = -1 }
				}
			}
			#省份总粮食产出
			every_owned_state = {
				set_variable = { this.state_food_production = 0 }
				add_to_variable = { this.state_food_production = this.resource@food }
			}
			if = {
				limit = { ROOT = { has_idea = AgriculturalCooperatives } }
				every_owned_state = { multiply_variable = { state_food_production = 1.3 } }
			}
			every_owned_state = {
				if = {
					limit = {
						check_variable = { this.state_food_production < state_food_need }
						check_variable = { this.infrastructure_level > 1 }
					}
					every_neighbor_state = {
						limit = {
							check_variable = { this.state_food_production > this.state_food_need }
							check_variable = { THIS.controller = PREV.controller }
						}
						add_to_variable = { PREV.state_food_neighber = 1 }
						if = {
							limit = {
								check_variable = { this.state_food_production > this.state_food_need }
								check_variable = { this.state_food_production > 20 }
								check_variable = { THIS.controller = PREV.controller }
							}
							add_to_variable = { PREV.state_food_neighber = 2 }
						}
					}
				}
			}
			#食物相邻补充
			every_owned_state = {
				if = {
					limit = {
						check_variable = { this.state_food_neighber > 0 }
					}
					set_variable = { this.state_food_neighber = 0 }
				}
			}
			#海产计算
			if = {
				limit = {
					check_variable = { this.fish_num > 0 }
				}
				set_variable = { this.fish_num = 0 }
			}
			every_owned_state = {
				if = {
					limit = {
						has_dynamic_modifier = { modifier = fish_production }
					}
					set_variable = { this.state_fish_num = 0 }
					add_to_variable = { this.state_fish_num = 3 }
					add_to_variable = { this.state_fish_num = building_level@dockyard }
					add_to_variable = { this.state_fish_num = building_level@dockyard }
					if = {
						limit = { PREV = { has_completed_focus = common_fishing_1 } }
						multiply_variable = { this.state_fish_num = 1.25 }
					}
					if = {
						limit = { PREV = { has_completed_focus = common_fishing_2 } }
						multiply_variable = { this.state_fish_num = 1.25 }
					}
					add_to_variable = { this.state_food_production = state_fish_num }
					add_to_variable = { PREV.fish_num = this.state_fish_num }
				}
			}
			##buff影响的食物产量
			if = {
				limit = { has_idea = PeoplesCommuni }
				every_owned_state = { multiply_variable = { state_fish_num = 0.8 } }
			}
			if = {
				limit = { has_idea = war_economy }
				every_owned_state = { multiply_variable = { state_fish_num = 0.8 } }
			}
			##来自加工食品
			if = {
				limit = { has_idea = PeoplesCommuni }
				every_owned_state = { multiply_variable = { state_fish_num = 0.8 } }
			}
			#粮食转运计算
			if = {
				limit = { check_variable = { foods_factory_modifier > 0 } }
				add_to_variable = { fish_num = this.foods_factory_modifier }
			}
			##饥荒条件
			every_owned_state = {
				limit = {
					check_variable = { this.state_food_production < this.state_food_need }
				}
				set_variable = { this.state_food_low_limit = state_food_need }
				multiply_variable = { this.state_food_low_limit = 0.6 }
			}
			##扣除进入储存的粮食
			every_owned_state = {
				if = {
					limit = { check_variable = { this.foods_to_save > 0 } }
					subtract_from_variable = { this.nation_food_cost = this.foods_to_save }
				}
			}
			##粮食满足的省份计算
			
			set_variable = { food = 0 }
			every_owned_state = {
				limit = {
					NOT = {
					check_variable = { this.state_food_production < this.state_food_low_limit }
					}
				}
				add_to_variable = { PREV.food = 1 }
			}
		#粮食库存计算
		set_variable = { food_num = 0 }
		every_owned_state = {
			add_to_variable = { PREV.food_num = state_food_production }
		}
		#粮食消耗计算
		set_variable = { nation_food_cost = 0 }
		every_owned_state = {
			add_to_variable = { PREV.nation_food_cost = state_food_need }
		}
			if = {
				limit = { check_variable = { total_state_food_neighber > 0 } }
				add_to_variable = { nation_food_cost = total_state_food_neighber }
			}
		set_variable = { nation_food_free = food_num }
		subtract_from_variable = { nation_food_free = nation_food_cost }
		#大饥荒
		if = {
			limit = {
				tag = PRC
				has_idea = PeoplesCommuni
				has_idea = GreatLeapForward_ideas
			}
			every_owned_state = {
				if = {
					limit = { has_state_flag = great_farmines }
					multiply_variable = { state_food_production = 0.2 }
				}
			}
		}
	}
}

news_event = {
	id = count.3
	hidden = yes
	mean_time_to_happen = {

	}

		}
	}
	option = {
		name = news.51.a
	#保健需求
			every_owned_state = {
				if = {
					limit = { check_variable = { state_health_need > 0 } }
					set_variable = { state_industry_health_need = 0 }
					set_variable = { state_civil_health_need = 0 }
				}
			}
			every_owned_state = {
				if = {
					limit = { has_state_category = rural }
					add_to_variable = { state_civil_health_need = 2 }
				}
				if = {
					limit = { has_state_category = town }
					add_to_variable = { state_civil_health_need = 3 }
				}
				if = {
					limit = { has_state_category = large_town }
					add_to_variable = { state_civil_health_need = 4 }
				}
				if = {
					limit = { has_state_category = city }
					add_to_variable = { state_civil_health_need = 5 }
				}
				if = {
					limit = { has_state_category = large_city }
					add_to_variable = { state_civil_health_need = 6 }
				}
				if = {
					limit = { has_state_category = metropolis }
					add_to_variable = { state_civil_health_need = 8 }
				}
				if = {
					limit = { has_state_category = megalopolis }
					add_to_variable = { state_civil_health_need = 8 }
				}
				else_if = {
					limit = { has_dynamic_modifier = { modifier = state_cold_modifier } }
					multiply_variable = { state_civil_health_need = 0.8 }

				add_to_variable = { state_health_need = state_civil_health_need }
			}
	#保健供给
			every_owned_state = {
				if = {
					limit = {
						hospital > 0
					}
					set_variable = { state_health_production = 0 }
					add_to_variable = { state_health_production = building_level@hospital }
					multiply_variable = { state_health_production = 4 }
				}
			}
			if = {
				limit = { has_idea = no_welfare }
				every_owned_state = {
					multiply_variable = { state_health_production = 0.2 }
				}
			}
			if = {
				limit = { has_idea = low_welfare }
				every_owned_state = {
					multiply_variable = { state_health_production = 0.6 }
				}
			}
			if = {
				limit = { has_idea = hight_welfare }
				every_owned_state = {
					multiply_variable = { state_health_production = 1.2 }
				}
			}
			if = {
				limit = { has_idea = welfare_state }
				every_owned_state = {
					multiply_variable = { state_health_production = 1.5 }
				}
			}
			if = {
				limit = { has_country_flag = medical_college }
				every_owned_state = {
					multiply_variable = { state_health_production = 1.2 }
				}
			}
			if = {
				limit = {
					has_country_flag = rural_medic
				}
				every_owned_state = {
					limit = {
						OR = {
							has_state_category = rural
							has_state_category = pastoral
							has_state_category = town
						}

				}
			}
