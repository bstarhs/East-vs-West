news_event = {  #识字率
	id = eco.12
	title = eco.12.t
	desc = eco.12.d
	picture = GFX_education
	mean_time_to_happen = { days = 30 }
	trigger = {
		NOT = {
			OR = {
			has_idea = low_education
			has_idea = mid_education
			}
		}
		check_variable = { this.education < 60 }
	}
	option = {
		name = eco.11.a
			add_popularity = {
				ideology = ROOT
				popularity = -0.04
			}
		if = {
			limit = {
				check_variable = { this.education < 25 }
			}
			add_ideas = low_education
		}
		if = {
			limit = {
				check_variable = { this.education > 25 }
				check_variable = { this.education < 60 }
			}
			add_ideas = mid_education
		}
	}
}
news_event = {
	id = eco.13
	title = eco.13.t
	desc = eco.13.d
	picture = GFX_education
	mean_time_to_happen = { days = 30 }
	trigger = {
		OR = {
			AND = {
				has_idea = low_education
				check_variable = { this.education > 25 }
			}
			AND = {
				has_idea = mid_education
				check_variable = { this.education > 60 }
			}
		}
	}
	option = {
		name = eco.11.a
			add_popularity = {
				ideology = ROOT
				popularity = 0.02
			}
		if = {
			limit = {
				check_variable = { this.education > 60 }
			}
			remove_ideas = low_education
			remove_ideas = mid_education
		}
		if = {
			limit = {
				check_variable = { this.education > 25 }
				check_variable = { this.education < 60 }
			}
			remove_ideas = low_education
			add_ideas = mid_education
		}
	}
}
