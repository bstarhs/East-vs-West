scripted_gui = {
	diplo_more_button = {
		window_name = "diplo_more_button"
		context_type = selected_country_context
		parent_window_token = selected_country_view
        visible = {
			#######################
			NOT = {
				tag = ROOT
			}
			OR = {
				AND = {
					tag = USA
					PREV = {
						OR = {
						has_country_flag = UsaAdvisorAllow
						has_country_flag = usa_arms_sell
						}
					}
				}
					
				AND = {
					tag = SOV
					PREV = {
						OR = {
						has_country_flag = sov_arms_sell
						has_government = communism
						}
					}
				}
				PREV = {
					OR = {
						is_in_faction_with = ROOT
						is_subject_of = ROOT
						is_puppet_of = ROOT
						is_guaranteed_by = ROOT
					}
				}
			}
		}
		effects = {		
			diplo_more_button_click = {
                set_temp_variable = { show = 1 }
                subtract_from_temp_variable = { show = show_diplo_more }
                set_variable = { show_diplo_more = show }
            }
        }
	}
}
