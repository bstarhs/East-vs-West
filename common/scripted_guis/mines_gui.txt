scripted_gui = {
	mines_button = {
		window_name = "mines_button_container"
		
		context_type = player_context
		ai_enabled = { always = yes }
		ai_test_interval = 10
		ai_check = {
			date > 1945.1.1
		}
		triggers = {
			mines_button_tips_visible = {
				NOT = { check_variable = { mines_button_tips > 0 } }
			}
			mines_button_tips_arrow_visible = {
				NOT = { check_variable = { mines_button_tips > 0 } }
			}
			alert_low_powers_visible = {
				any_owned_state = { check_variable = { state_power_production < state_power_need } }
			}
			alert_foods_lacking_visible = {
				any_owned_state = { check_variable = { this.state_food_production < this.state_food_low_limit } }
			}
		}
		effects = {		
			mines_button_click = {
                set_temp_variable = { show = 1 }
                subtract_from_temp_variable = { show = show_mines }
                set_variable = { show_mines = show }
                set_variable = { show_mines_list = 1 }
                set_variable = { mines_button_tips = 1 }
            }
			alert_low_powers_click = {
				random_owned_state = {
					limit = { check_variable = { state_power_production < state_power_need } }
					root = { goto_state = prev }
				}
            }
			alert_foods_lacking_click = {
				random_owned_state = {
					limit = { check_variable = { this.state_food_production < this.state_food_low_limit } }
					root = { goto_state = prev }
				}
            }
        }
	}
}
